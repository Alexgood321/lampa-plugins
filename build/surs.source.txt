            raw: def.raw || previous.raw || def,
            screen: def.screen !== undefined ? def.screen : previous.screen
        };
        saveDiscoveredToStorage();
        return true;
    }

    var discoveredSaveTimer = null, discoveredSaveRevision = 0, discoveredSavePending = false;
    function flushDiscoveredToStorage() {
        if (!discoveredSavePending) return;
        discoveredSavePending = false;
        if (discoveredSaveTimer !== null) {
            clearTimeout(discoveredSaveTimer);
            discoveredSaveTimer = null;
        }
        try {
            if (typeof Lampa === 'undefined' || !Lampa.Storage) {
                discoveredSavePending = true;
                return;
            }
            var list = [];
            for (var key in window.SURS_EXTERNAL_SHELVES) {
                var it = window.SURS_EXTERNAL_SHELVES[key];
                if (isAnonymousShelf(it)) continue;
                list.push({
                    id: it.id, title: it.title, cat: it.cat,
                    desc: it.desc, def: it.def
                });
            }
            var existing = Lampa.Storage.get(STORAGE_DISCOVERED_KEY, []);
            if (JSON.stringify(existing) !== JSON.stringify(list)) {
                Lampa.Storage.set(STORAGE_DISCOVERED_KEY, list);
                window.SURS_PERFORMANCE.discovered_writes =
                    (window.SURS_PERFORMANCE.discovered_writes || 0) + 1;
            }
        } catch(e) {
            discoveredSavePending = true;
        }
    }
    function saveDiscoveredToStorage() {
        discoveredSavePending = true;
        var revision = ++discoveredSaveRevision;
        if (discoveredSaveTimer !== null) clearTimeout(discoveredSaveTimer);
        discoveredSaveTimer = setTimeout(function() {
            if (revision === discoveredSaveRevision) flushDiscoveredToStorage();
        }, 400);
    }
    try {
        if (typeof window.addEventListener === 'function')
            window.addEventListener('pagehide', flushDiscoveredToStorage);
        if (typeof document !== 'undefined' && document.addEventListener)
            document.addEventListener('visibilitychange', function() {
                if (document.hidden) flushDiscoveredToStorage();
            });
    } catch(e) {}

    function loadDiscoveredFromStorage() {
        if (discoveredLoadedOnce) return;
        try {
            if (typeof Lampa === 'undefined' || !Lampa.Storage) return;
            var saved = Lampa.Storage.get(STORAGE_DISCOVERED_KEY, []);
            if (Array.isArray(saved)) {
                saved.forEach(function(item) {
                    if (item && item.id && !isAnonymousShelf(item) && !window.SURS_EXTERNAL_SHELVES[item.id]) {
                        window.SURS_EXTERNAL_SHELVES[item.id] = item;
                    }
                });
            }
            discoveredLoadedOnce = true;
        } catch(e) {}
    }

    function patchContentRows() {
        try {

                    var origAdd = Lampa.ContentRows.add;

                        // Keep native errors, return values and the original row object.

                        try {
                            if (rowDef && typeof rowDef === 'object' && typeof rowDef.call === 'function' && rowDef.name) {
                                registerDiscoveredShelf({
                                    id: 'ext_' + String(rowDef.name),
                                    title: rowDef.title || rowDef.name || 'Сторонняя полка',
                                    cat: 'plugin',
                                    desc: 'Lampa ContentRow (' + rowDef.name + ')',
                                    call: rowDef.call,
                                    screen: rowDef.screen,
                                    raw: rowDef
                                });
                            }
                        } catch(e) {
                            console.error('[SURS Hub] Ошибка перехвата ContentRows.add:', e);
                        }
                        return result;
                    };
                    wrappedAdd.__sursWrapped = true;
                    Lampa.ContentRows.add = wrappedAdd;
                    Lampa.ContentRows.__surs_patched = true;
                }
            }
        } catch(e) {}
    }

    function openGenre(genre) {
        var hasMovie = movieGenres.some(function(g) { return g.id === genre.id; });
        var hasTv = tvGenres.some(function(g) { return g.id === genre.id; });

        var typeChoices = [];
        if (hasMovie) typeChoices.push({ title: Lampa.Lang.translate('menu_movies'), value: 'movie' });
        if (hasTv) typeChoices.push({ title: Lampa.Lang.translate('menu_tv'), value: 'tv' });

        var title = Lampa.Lang.translate(genre.title) || genre.title;

        if (typeChoices.length === 1) {
            showGenreSortMenu(genre, typeChoices[0].value, title);
            return;
        }

        Lampa.Select.show({
            title: title,
            items: typeChoices.map(function(tc) {
                return {
                    title: tc.title,
                    action: function() { showGenreSortMenu(genre, tc.value, title + ' · ' + tc.title); }
                };
            }),
            onSelect: function(i) { i.action(); },
            onBack: function() { Lampa.Controller.toggle('content'); }
        });
    }

    function getAllAvailableShelves() {
        loadDiscoveredFromStorage();
        var map = {};
        var list = [];

        function add(item) {
            if (!item || !item.id) return;
            if (!map[item.id]) {
                map[item.id] = true;
                list.push(item);
            }
        }

        SURS_BASE_SHELVES.forEach(add);
        LAMPA_CORE_SHELVES.forEach(add);

        for (var k in window.SURS_EXTERNAL_SHELVES) {
            add(window.SURS_EXTERNAL_SHELVES[k]);
        }

        return list;
    }

    function getShelfCategory(id) {
        var all = getAllAvailableShelves();
        for (var i = 0; i < all.length; i++) {
            if (all[i].id === id) return all[i].cat || 'surs';
        }
        if (id.indexOf('core_') === 0) return 'core';
        if (id.indexOf('trakt_') === 0) return 'trakt';
        if (id.indexOf('ext_') === 0) return 'plugin';
        if (id.indexOf('cr_') === 0) return 'plugin';
        return 'surs';
    }

    function getShelfDesc(id) {
        var all = getAllAvailableShelves();
        for (var i = 0; i < all.length; i++) {
            if (all[i].id === id) return all[i].desc || '';
        }
        return '';
    }

    function runAutoDiscovery(notifyUser) {
        loadDiscoveredFromStorage();
        var beforeCount = Object.keys(window.SURS_EXTERNAL_SHELVES).length;

        // 1. Не вызываем ContentRows.call ради обнаружения: этот метод исполняет
        // сторонние загрузчики, а их функции не имеют стабильных идентификаторов.
        // Новые полки регистрирует перехватчик ContentRows.add; ранее загруженные
        // плагины могут зарегистрироваться через window.surs_register_shelf.

        // 2. Проверяем Trakt
        try {
            var traktToken = (typeof Lampa !== 'undefined' && Lampa.Storage) ? 
                (Lampa.Storage.get('trakt_token') || Lampa.Storage.get('trakt')) : null;
            var hasTraktObj = (typeof Lampa !== 'undefined' && typeof Lampa.Trakt !== 'undefined') || 
                              (typeof window !== 'undefined' && typeof window.TraktTV !== 'undefined');
            if (traktToken || hasTraktObj) {
                registerDiscoveredShelf({
                    id: 'trakt_continue',
                    title: 'Trakt: Смотреть дальше',
                    cat: 'trakt',
                    desc: 'Недосмотренные фильмы и сериалы Trakt.tv',
                    def: false
                });
                registerDiscoveredShelf({
                    id: 'trakt_watchlist',
                    title: 'Trakt: Хочу посмотреть',
                    cat: 'trakt',
                    desc: 'Ваш список к просмотру (Watchlist) из Trakt',
                    def: false
                });
                registerDiscoveredShelf({
                    id: 'trakt_recommendations',
                    title: 'Trakt: Рекомендации',
                    cat: 'trakt',
                    desc: 'Персональные рекомендации Trakt',
                    def: false
                });
                registerDiscoveredShelf({
                    id: 'trakt_trending',
                    title: 'Trakt: В тренде',
                    cat: 'trakt',
                    desc: 'Тренды кино и сериалов Trakt',
                    def: false
                });
            }
        } catch(e) {}

        saveDiscoveredToStorage();

        var all = getAllAvailableShelves();
        var afterCount = Object.keys(window.SURS_EXTERNAL_SHELVES).length;
        var diff = afterCount - beforeCount;

        if (notifyUser && typeof Lampa !== 'undefined' && Lampa.Noty) {
            var msg = 'Ресерч завершён: доступно ' + all.length + ' полок';
            if (diff > 0) msg += ' (найдено новых: ' + diff + ')';
            Lampa.Noty.show(msg);
        }

        return {
            total: all.length,
            discovered: afterCount,
            newCount: diff
        };
    }

    function activeShelfScope() {
        try {
            var activity = Lampa.Activity && Lampa.Activity.active && Lampa.Activity.active();
            var source = activity && activity.source;
            if (source === 'SURS NEW') return 'NEW';
            if (source === 'SURS RUS') return 'RUS';
        } catch(e) {}
        return 'MAIN';
    }

    function getDefaultSectionShelves(scope) {
        var name = normalizeShelfScope(scope), config = SURS_SECTION_PROFILES[name];
        if (!config) return getDefaultShelvesList();
        var catalog = getAllAvailableShelves();
        var known = {};
        catalog.forEach(function(shelf) { if (shelf && shelf.id) known[shelf.id] = shelf; });
        return config.defaults.map(function(id) {
            var item = known[id] || { id: id, title: id, cat: 'surs' };
            return {
                id: id, title: item.title, cat: item.cat,
                desc: item.desc, enabled: config.enabled.indexOf(id) !== -1
            };
        });
    }

    function executeBulletproofReset(scope) {
        // Сохраняем непосредственно новый профиль. Промежуточная запись [] и
        // removeItem могли конфликтовать с кэшем Lampa.Storage.
        var name = normalizeShelfScope(scope);
        var cleanDefaults = getDefaultSectionShelves(name);
        if (!saveShelves(cleanDefaults, name)) return null;
        if (name === 'MAIN') Lampa.Storage.set('surs_plugins_default_off_v2', 'true');
        return cleanDefaults;
    }

    function refreshSursHome(scope) {
        setTimeout(function() {
            try {
                var name = normalizeShelfScope(scope);
                var profile = SURS_SECTION_PROFILES[name];
                var activity = {
                    title: shelfScopeTitle(name), component: 'main',
                    source: profile ? profile.source : 'SURS', page: 1
                };
                if (Lampa.Activity && typeof Lampa.Activity.replace === 'function') {
                    Lampa.Activity.replace(activity);
                } else if (Lampa.Activity && typeof Lampa.Activity.push === 'function') {
                    Lampa.Activity.push(activity);
                }
            } catch(err) {
                console.error('[SURS] Ошибка обновления главного экрана после сброса:', err);
            }

    function migrateLegacyFactoryOrder(saved) {
        if (!Array.isArray(saved) || saved.length !== 22) return saved;
        var legacyIds = [
            'nav_buttons', 'core_continue', 'streaming_russian', 'streaming_global',
            'genres_icons', 'trend_movies', 'trend_tv', 'upcoming_episodes',
            'new_movies', 'new_tv', 'rus_movies', 'rus_tv',
            'stream_kinopoisk', 'stream_premier', 'stream_netflix', 'stream_apple',
            'turkish_tv', 'korean_tv', 'genre_comedy', 'genre_action',
            'genre_scifi', 'period_00s'
        ];
        var lastFactoryIds = [
            'nav_buttons', 'core_continue', 'new_movies', 'new_tv',
            'trend_movies', 'trend_tv', 'upcoming_episodes', 'rus_movies',
            'rus_tv', 'genres_icons', 'streaming_russian', 'streaming_global',
            'stream_kinopoisk', 'stream_premier', 'stream_netflix', 'stream_apple',
            'turkish_tv', 'korean_tv', 'genre_comedy', 'genre_action',
            'genre_scifi', 'period_00s'
        ];
        var originalFactory = saved.every(function(item, index) {
            return item && item.id === legacyIds[index] && item.enabled === true;
        });
        var lastFactory = saved.every(function(item, index) {
            return item && item.id === lastFactoryIds[index] && item.enabled === (index < 12);
        });
        if (!originalFactory && !lastFactory) return saved;
        var defaults = getDefaultShelvesList();
        return saveShelves(defaults) ? defaults : saved;
    }

    function getSavedShelves(scope) {
        var name = normalizeShelfScope(scope);
        var profile = SURS_SECTION_PROFILES[name];
        var saved = null;
        try {
            saved = Lampa.Storage.get(profile ? profile.key : STORAGE_SHELVES_KEY, null);
            if (!profile) saved = migrateLegacyFactoryOrder(saved);
        } catch(e) {}

        if (Array.isArray(saved)) {
            var seen = {};
            var unique = [];
            var hadSmart = !profile && saved.some(function(item) {
                return item && item.id === 'smart_us_new';
            });
            saved.forEach(function(s) {
                if (s && s.id && !seen[s.id] && (!isAnonymousShelf(s) || s.enabled)) {
                    seen[s.id] = true;
                    // Retired Smart US must never be rendered after the upgrade.
                    if (!profile && s.id === 'smart_us_new') return;
                    // Keep existing ID, enabled state and order for other shelves.
                    unique.push(!profile && s.id === 'kp_new_8' ?
                        Object.assign({}, s, {title: 'Новинки США · КП 7,5+'}) : s);
                }
            });
            // If a saved MAIN profile contained only the retired Smart shelf,
            // rebuild a minimal navigable home with Hot; an explicitly empty
            // profile without Smart retains its existing navigation-only fallback.
            if (!unique.length) {
                if (profile) return getDefaultSectionShelves(name);
                var fallback = hadSmart ? getDefaultShelvesList().slice(0, 2) :
                    [getDefaultShelvesList()[0]];
                if (hadSmart) saveShelves(fallback, 'MAIN');
                return fallback;
            }
            // Hot Discovery is the only primary US shelf. Add it on migration
            // without changing the saved order; preserve explicit Hot opt-outs.
            if (!profile && seen.nav_buttons && !seen.hot_us_new &&
                !Lampa.Storage.get('surs_hot_default_v402', false)) {
                var navIndex = unique.findIndex(function(x) { return x.id === 'nav_buttons'; });
                unique.splice(navIndex + 1, 0, {
                    id: 'hot_us_new', title: 'Горячие новинки США',
                    enabled: true, cat: 'surs'
                });
            }
            // Persist removal of retired Smart US; retain every other row and
            // explicit on/off choice, including those for Hot Discovery.
            if (hadSmart) saveShelves(unique, 'MAIN');
            return unique;
        }

        return getDefaultSectionShelves(name);
    }

    function saveShelves(list, scope) {
        var profile = SURS_SECTION_PROFILES[normalizeShelfScope(scope)];
        var key = profile ? profile.key : STORAGE_SHELVES_KEY;
        try {
            if (!Array.isArray(list)) throw new Error('Invalid shelf list');
            var saveError = null;
            // Lampa.Storage.set(name, value, nolisten, callerror) reports quota
            // errors through the fourth argument; the return value is void.
            Lampa.Storage.set(key, list, false, function(err) {
                saveError = err || new Error('Storage error');
            });
            if (saveError) throw saveError;
            var stored = Lampa.Storage.get(key, null);
            if (JSON.stringify(stored) !== JSON.stringify(list))
                throw new Error('Storage read-back mismatch');
            // An explicit saved MAIN edit can remove Hot without re-adding it.
            if (!profile) Lampa.Storage.set('surs_hot_default_v402', true);
            return true;
        } catch(err) {
            console.error('[SURS] Ошибка сохранения полок:', key, err);
            if (Lampa.Noty && typeof Lampa.Noty.show === 'function') {
                Lampa.Noty.show('Не удалось сохранить полки. Изменения остались в редакторе.');
            }
            return false;
        }
    }

    function hotLoad(owner,done){
        var old=hotReadCached();
        if(old){done(old);if(Date.now()-hotLastTry<3*3600000)return;}
        else if(Date.now()-hotLastTry<30000)return done(null);
        if(hotRequesting){if(!old)hotCallbacks.push(done);return;}
        hotRequesting=true;hotLastTry=Date.now();
        if(!old)hotCallbacks.push(done);
        var finished=false,timeout=setTimeout(function(){finish(null,'timeout');},3500);
        function finish(feed,error){
            if(finished)return;
            finished=true;clearTimeout(timeout);hotRequesting=false;
            if(hotValid(feed)){
                try{Lampa.Storage.set(HOT_CACHE_KEY,feed);}catch(e){}
                window.SURS_PERFORMANCE.hot_us={
                    status:'ready',generated_at:feed.generated_at,
                    algorithm:feed.algorithm||'unknown',
                    source:feed.sources||{},diagnostics:feed.diagnostics||{},
                    top10:feed.items.slice(0,10).map(function(c){
                        return {title:c.title||c.name,score:c.surs_hot_score,
                            quality:c.surs_hot_quality,heat:c.surs_hot_heat,
                            release:c.surs_hot_premiere_date,kind:c.surs_hot_premiere_kind};
                    })
                };
            }else{
                var fallback=old||hotReadLegacyCached();
                feed=fallback;
                window.SURS_PERFORMANCE.hot_us={
                    status:fallback?'cached':'unavailable',
                    algorithm:fallback?fallback.algorithm||'unknown':'unknown',
                    error:error||'invalid_feed',
                    generated_at:fallback?fallback.generated_at:null,
                    diagnostics:fallback?fallback.diagnostics||{}:{}
                };
            }
            hotCallbacks.splice(0).forEach(function(fn){try{fn(feed);}catch(e){}});
        }
        try{
            var network=owner&&owner.network&&typeof owner.network.silent==='function'?
                owner.network:new Lampa.Reguest();
            if(!network||typeof network.silent!=='function')
                return finish(null,'network_unavailable');
            network.silent(HOT_US_URL+'?v='+Math.floor(Date.now()/3600000),
                function(json){
                    try{if(typeof json==='string')json=JSON.parse(json);}
                    catch(e){json=null;}
                    finish(json,'invalid_response');
                },function(){finish(null,'request_failed');});
        }catch(e){finish(null,'network_error');}
    }

    var US8_CATALOG_URL = 'surs_us_new_8';
    var US8_TYPES = [
        { id: 'movie', date: 'primary_release_date', sort: 'release_date.desc',
          votes: 300, weight: 400, blocked: '16,99,10770' },
        { id: 'tv', date: 'first_air_date', sort: 'first_air_date.desc',
          votes: 150, weight: 200, blocked: '16,99,10763,10764,10766,10767' }
    ];
    function us8Dates() {
        var now = new Date();
        var end = now.toISOString().slice(0, 10);
        now.setFullYear(now.getFullYear() - 1);
        return { start: now.toISOString().slice(0, 10), end: end };
    }
    function us8Query(type, dates, page) {
        return 'discover/' + type.id +
            '?with_origin_country=US&with_original_language=en' +
            '&include_adult=false&without_genres=' + type.blocked +
            '&vote_average.gte=8&vote_count.gte=' + type.votes +
            '&' + type.date + '.gte=' + dates.start +
            '&' + type.date + '.lte=' + dates.end +
            '&sort_by=' + type.sort + '&page=' + page;
    }
    function us8CardDate(card) {
        return card.release_date || card.first_air_date || '';
    }
    // Bayesian score: downweight high ratings until enough votes accumulate.
    // The displayed TMDB rating is never changed.
    function us8WeightedRating(item, type) {
        var votes = Number(item.vote_count) || 0;
        var rating = Number(item.vote_average) || 0;
        return (votes * rating + type.weight * 7) / (votes + type.weight);
    }
    function us8Compare(a, b) {
        var da = us8CardDate(a), db = us8CardDate(b);
        if (da !== db) return da > db ? -1 : 1;
        if (a.surs_weighted_score !== b.surs_weighted_score)
            return b.surs_weighted_score - a.surs_weighted_score;
        return Number(b.vote_count) - Number(a.vote_count);
    }
    function us8Collect(json, type, dates, seen, cards) {
        if (!json || !Array.isArray(json.results)) return;
        var blocked = type.blocked.split(',');
        json.results.forEach(function(item) {
            if (!item || !item.id || !item.poster_path ||
                !item.overview || !String(item.overview).trim() ||
                item.original_language !== 'en' || item.adult === true ||
                Number(item.vote_average) < 8 || Number(item.vote_count) < type.votes) return;
            var weightedScore = us8WeightedRating(item, type);
            if (weightedScore < 7.5) return;
            var date = type.id === 'tv' ? item.first_air_date : item.release_date;
            if (!date || date < dates.start || date > dates.end) return;
            if (type.id === 'tv' && (!Array.isArray(item.origin_country) ||
                                    item.origin_country.indexOf('US') < 0)) return;
            if (Array.isArray(item.genre_ids) && item.genre_ids.some(function(id) {
                return blocked.indexOf(String(id)) >= 0;
            })) return;
            var key = type.id + ':' + item.id;
            if (seen[key]) return;
            seen[key] = true;
            item.source = 'tmdb';
            item.type = type.id;
            item.media_type = type.id;
            item.surs_weighted_score = weightedScore;
            cards.push(item);
        });
    }
    function us8RequestParams(params) {
        var copy = Object.assign({}, params || {});
        delete copy.page;
        delete copy.url;
        delete copy.source;
        delete copy._sursUS8Session;
        return copy;
    }
    function us8CatalogSession(owner, params) {
        var dates = us8Dates();
        var requestParams = us8RequestParams(params);
        var seen = {}, cards = [], queue = [], working = false;
        var states = US8_TYPES.map(function(type) {
            return { type: type, next: 1, total: 0, boundary: null, exhausted: false };
        });
        function run() {
            if (working || !queue.length) return;
            working = true;
            var req = queue.shift(), page = req.page, target = page * 20;
            var settled = false;
            var deadline = setTimeout(complete, 8500);
            function complete() {
                if (settled) return;
                settled = true;
                clearTimeout(deadline);
                cards.sort(us8Compare);
                var results = cards.slice((page - 1) * 20, target);
                var morePossible = states.some(function(st) { return !st.exhausted; });
                var totalPages = Math.max(1, Math.ceil(cards.length / 20));
                if (morePossible) {
                    var rawPages = states.reduce(function(n, st) {
                        return n + (st.total || Math.max(2, st.next));
                    }, 0);
                    totalPages = Math.max(totalPages, page + 1,
                        Math.min(100, rawPages));
                }
                // Lampa's native category pagination must see total_pages on page 1.
                if (results.length) {
                    req.ok({
                        title: 'Новинки США 8+', url: US8_CATALOG_URL,
                        source: 'SURS', page: page, total_pages: totalPages,
                        results: results
                    });
                } else if (req.fail) req.fail();
                working = false;
                run();
            }
            function next() {
                if (settled) return;
                cards.sort(us8Compare);
                var cutoff = cards[target - 1] ? us8CardDate(cards[target - 1]) : null;
                var needs = states.filter(function(st) {
                    return !st.exhausted &&
                        (!cutoff || !st.boundary || st.boundary >= cutoff);
                });
                if (!needs.length) return complete();
                var pending = needs.length;
                needs.forEach(function(st) {
                    var rawPage = st.next++;
                    var responded = false;
                    function receive(json) {
                        if (responded) return;
                        responded = true;
                        if (!settled) {
                            var raw = json && Array.isArray(json.results) ? json.results : [];
                            if (json && typeof json.total_pages === 'number')
                                st.total = Math.min(500, Math.max(1, json.total_pages));
                            st.boundary = raw.length ? us8CardDate(raw[raw.length - 1]) : '';
                            us8Collect(json, st.type, dates, seen, cards);
                            if (!raw.length || (st.total && rawPage >= st.total) ||
                                ((!json || typeof json.total_pages !== 'number') && raw.length < 20))
                                st.exhausted = true;
                        }
                        if (--pending === 0) next();
                    }
                    try {
                        owner.get(us8Query(st.type, dates, rawPage), requestParams,
                            receive, function() { receive(null); });
                    } catch(e) { receive(null); }
                });
            }
            next();
        }
        return {
            load: function(page, ok, fail) {
                var number = Math.max(1, Math.min(500, Number(page) || 1));
                queue.push({ page: number, ok: ok, fail: fail });
                run();
            }
        };
    }

    var kp8DependencyStatus = 'idle';
    var kp8DependencyFailedAt = 0;
    var kp8DependencyWaiters = [];
    var kp8DependencyError = '';
    function kp8RegisteredSource() {
        var source = Lampa.Api && Lampa.Api.sources && Lampa.Api.sources.KP;
        return source && typeof source.list === 'function' ? source : null;
    }
    function kp8LoadDependency(done) {
        var source = kp8RegisteredSource();
        if (source) {
            kp8DependencyStatus = 'ready';
            done(source);
            return;
        }
        if (typeof document === 'undefined' || !document.head ||
            typeof document.createElement !== 'function') {
            kp8DependencyError = 'no_document';
            done(null);
            return;
        }
        kp8DependencyWaiters.push(done);
        if (kp8DependencyStatus === 'loading') return;
        if (kp8DependencyStatus === 'failed') {
            // Only retry the original pinned dependency after a cooldown.
            if(Date.now()-kp8DependencyFailedAt<15*60000){
                var failedWaiters=kp8DependencyWaiters.splice(0);
                failedWaiters.forEach(function(fn){try{fn(null);}catch(e){}});
                return;
            }
            kp8DependencyStatus='idle';
        }

        kp8DependencyStatus = 'loading';
        kp8DependencyError = '';
        var settled = false, script = null;
        var timer = setTimeout(function() { finish(null, 'load_timeout'); }, 6500);
        function finish(found, error) {
            if (settled) return;
            settled = true;
            clearTimeout(timer);
            kp8DependencyStatus = found ? 'ready' : 'failed';
            kp8DependencyFailedAt=found?0:Date.now();
            kp8DependencyError = found ? '' : (error || 'not_registered');
            window.SURS_PERFORMANCE.kp8_dependency = {
                status: kp8DependencyStatus,
                error: kp8DependencyError,
                url: KP8_SOURCE_URL
            };
            if (script) {
                script.onload = null;
                script.onerror = null;
            }
            var callbacks = kp8DependencyWaiters.splice(0);
            callbacks.forEach(function(fn) {
                try { fn(found); } catch(e) {
                    console.warn('[SURS] KP dependency callback failed:', e);
                }
            });
        }
        function waitForRegistration() {
            if (settled) return;
            var registered = kp8RegisteredSource();
            if (registered) return finish(registered, null);
            // KP source can wait for app.ready before registering itself.
            setTimeout(waitForRegistration, 150);
        }
        try {
            script = document.createElement('script');
            script.async = true;
            script.src = KP8_SOURCE_URL;
            script.onload = waitForRegistration;
            script.onerror = function() { finish(null, 'script_download_failed'); };
            document.head.appendChild(script);
        } catch(e) {
            finish(null, 'script_injection_failed');
        }
    }
    function kp8FindSource(done) {
        kp8LoadDependency(function(source) {
            done(source ? { source: source, country: KP8_US_COUNTRY_ID } : null);
        });
    }

    }

        window.__SURS_KP_LAST_NOTICE = reason;

    function kp8Query(type, country, years) {
        return 'api/v2.2/films?order=YEAR&countries=' + encodeURIComponent(country) +
            '&ratingFrom=' + KP8_MIN_RATING + '&ratingTo=10&yearFrom=' + years.from +
            '&yearTo=' + years.to + '&type=' + type.sourceType;
    }

    function kp8Fetch(type, page, ready, onFinish, years) {
        var responded = false;
        var timeout = setTimeout(function() { finish(null, 'timeout'); }, 11000);
        function finish(data, reason) {
            if (responded) return;
            responded = true;
            clearTimeout(timeout);
            onFinish(data, reason || '');
        }
        try {
            ready.source.list({
                url: 'filter',
                filter_url: kp8Query(type, ready.country, years || kp8Years()),
                page: page
            }, function(json) { finish(json, null); },
            function() { finish(null, 'request_failed'); });
        } catch(e) { finish(null, 'exception'); }
    }

    function kp8Date(card) {
        var date = card.release_date || card.first_air_date || '';
        return String(date).slice(0, 10);
    }
    function kp8Compare(a, b) {
        var dateA = kp8Date(a), dateB = kp8Date(b);
        if (dateA !== dateB) return dateA > dateB ? -1 : 1;
        if (a.kp_rating !== b.kp_rating) return b.kp_rating - a.kp_rating;
        return (Number(b.vote_count) || 0) - (Number(a.vote_count) || 0);
    }
    function kp8Collect(json, type, seen, result) {
        if (!json || !Array.isArray(json.results)) return;
        var years = kp8Years();
        json.results.forEach(function(card) {
            if (!card || !card.id || card.adult === true ||
                card.type !== type.id || !isFinite(Number(card.kp_rating)) ||
                Number(card.kp_rating) < KP8_MIN_RATING ||
                (Number(card.vote_count) > 0 && Number(card.vote_count) < type.votes)) return;
            var year = Number(kp8Date(card).slice(0, 4));
            if (!year || year < years.from || year > years.to) return;
            var genres = Array.isArray(card.genres) ? card.genres : [];
            if (genres.some(function(g) {
                return /аниме|мультфильм|анимац|короткометраж|документальн|реалити|ток.шоу|телешоу|для взрослых/i.test(g.name || '');
            })) return;
            var countries = Array.isArray(card.production_countries) ? card.production_countries : [];
            if (countries.length && !countries.some(function(c) { return c.name === 'США'; })) return;
            var key = String(card.id);
            if (seen[key]) return;
            seen[key] = true;
            card.kp_rating = Number(card.kp_rating);
            card.source = 'KP';
            result.push(card);
        });
    }

    function kp8Preview(callback) {
        var cachedPreview = readSursPreviewCache(KP8_PREVIEW_CACHE_KEY);
        var metrics = {
            source: 'loading', status: 'loading', pages_requested: 0,
            pages_completed: 0, pages_failed: 0, raw_candidates: 0,
            matched: 0, timeout: false, failed_reason: '',
            stop_reason: '', max_pages_per_type: KP8_PREVIEW_PAGES_PER_TYPE,
            target: KP8_PREVIEW_TARGET, cache_hit: !!cachedPreview,
            pages_movie: 0, pages_series: 0, pages_mini_series: 0
        };
        window.SURS_PERFORMANCE.kp8_preview = metrics;
        if (cachedPreview) {
            callback({
                title: 'Новинки США · КП 7,5+', url: KP8_CATALOG_URL,
                source: 'SURS', page: 1,
                total_pages: cachedPreview.total_pages,
                results: cachedPreview.results,
                params: { items: { view: 20, mapping: 'line' } }
            });
            metrics.source = 'cache';
            metrics.status = 'cached';
            if (cachedPreview.age < 4 * 60 * 60 * 1000) {
                metrics.matched = cachedPreview.results.length;
                metrics.stop_reason = 'cache';
                return;
            }
        }
        kp8FindSource(function(ready) {
            if (!ready) {
                metrics.source = 'missing';
                metrics.status = 'missing_source';
                metrics.failed_reason = kp8DependencyError || 'source_not_registered';
                metrics.dependency_status = kp8DependencyStatus;
                kp8Notice('missing_source', 'КП 7,5+: не удалось загрузить источник Кинопоиска (' + metrics.failed_reason + ').');
                if (!cachedPreview) callback({ results: [] });
                return;
            }
            metrics.source = 'KP';
            var seen = {}, cards = [], finished = false, round = 0;
            var states = KP8_TYPES.map(function(type, index) {
                return { type: type, next: 1, last: KP8_PREVIEW_PAGES_PER_TYPE,
                    metric: index === 0 ? 'pages_movie' :
                        (index === 1 ? 'pages_series' : 'pages_mini_series') };
            });
            // Keep the KP shelf within its existing 8s render watchdog so the
            // subsequent TMDB / Continue Watching rows are not blocked.
            var deadline = setTimeout(function() {
                metrics.timeout = true;
                metrics.stop_reason = 'timeout';
                deliver();
            }, 6900);

            function deliver() {
                if (finished) return;
                finished = true;
                clearTimeout(deadline);
                cards.sort(kp8Compare);
                metrics.matched = cards.length;
                if (cards.length) {
                    metrics.status = 'ready';
                    window.__SURS_KP_LAST_NOTICE = '';
                } else if (metrics.pages_failed === metrics.pages_requested && metrics.pages_requested) {
                    metrics.status = 'api_error';
                    kp8Notice('api_error', 'КП 7,5+: источник KP не отвечает. Проверьте его работу в Lampa.');
                } else if (metrics.timeout && metrics.pages_failed) {
                    metrics.status = 'timeout';
                    kp8Notice('timeout', 'КП 7,5+: превышено время ожидания ответа источника KP.');
                } else if (metrics.raw_candidates) {
                    metrics.status = 'filtered';
                    kp8Notice('filtered', 'КП 7,5+: API ответил, но по фильтрам США и рейтингу 8+ подходящих новинок нет.');
                } else {
                    metrics.status = 'empty_api';
                    kp8Notice('empty_api', 'КП 7,5+: источник вернул пустую подборку. Проверьте фильтры и доступность API.');
                }
                var result = {
                    title: 'Новинки США · КП 7,5+', url: KP8_CATALOG_URL,
                    source: 'SURS', page: 1, total_pages: cards.length ? 2 : 1,
                    results: cards.slice(0, 60),
                    params: { items: { view: 20, mapping: 'line' } }
                };
                saveSursPreviewCache(KP8_PREVIEW_CACHE_KEY, result);
                if (!cachedPreview) callback(result);
            }

            function nextBatch() {
                if (finished) return;
                if (round > 0 && cards.length >= KP8_PREVIEW_TARGET) {
                    metrics.stop_reason = 'target';
                    return deliver();
                }
                var jobs = [];
                // First query only page 1 per type to learn total_pages.
                // Later rounds may request two pages per type concurrently.
                var batchPages = round === 0 ? 1 : KP8_PREVIEW_BATCH_PAGES;
                states.forEach(function(state) {
                    for (var n = 0; n < batchPages &&
                         state.next <= state.last; n++) {
                        jobs.push({ state: state, page: state.next++ });
                    }
                });
                if (!jobs.length) {
                    metrics.stop_reason = 'exhausted';
                    return deliver();
                }
                round++;
                var pending = jobs.length;
                jobs.forEach(function(job) {
                    metrics.pages_requested++;
                    kp8Fetch(job.state.type, job.page, ready, function(json, failure) {
                        if (finished) return;
                        metrics.pages_completed++;
                        metrics[job.state.metric]++;
                        if (!json || !Array.isArray(json.results)) {
                            metrics.pages_failed++;
                            if (failure) metrics.failed_reason = failure;
                            job.state.last = Math.min(job.state.last, job.page);
                        } else {
                            metrics.raw_candidates += json.results.length;
                            kp8Collect(json, job.state.type, seen, cards);
                            if (typeof json.total_pages === 'number' &&
                                isFinite(json.total_pages)) {
                                job.state.last = Math.min(job.state.last,
                                    Math.max(1, Math.floor(json.total_pages)));
                            } else if (json.results.length < 20) {
                                job.state.last = Math.min(job.state.last, job.page);
                            }
                            if (!json.results.length)
                                job.state.last = Math.min(job.state.last, job.page);
                        }
                        if (--pending === 0) nextBatch();
                    });
                });
            }
            nextBatch();
        });
    }

    function kp8CatalogSession() {
        // Search current and preceding years separately so YEAR ordering
        // cannot suppress all the previous year's candidates.
        var ready = null, queued = [], working = false;
        var seen = {}, cards = [], ordered = [], orderedIds = {};
        var range = kp8Years(), states = [];
        [range.to, range.from].forEach(function(year) {
            KP8_TYPES.forEach(function(type) {
                states.push({
                    type: type, year: year, next: 1,
                    total: KP8_CATALOG_PAGES_PER_YEAR, exhausted: false
                });
            });
        });
        var metrics = { pages_requested: 0, pages_completed: 0,
            matched: 0, failed: 0, timeout: false };
        window.SURS_PERFORMANCE.kp8_catalog = metrics;

        function run() {
            if (working || !queued.length) return;
            working = true;
            var job = queued.shift(), page = job.page, completed = false;
            var deadline = setTimeout(function() {
                metrics.timeout = true;
                finish();
            }, 14500);
            function freeze() {
                cards.sort(kp8Compare);
                cards.forEach(function(card) {
                    if (!orderedIds[card.id]) {
                        orderedIds[card.id] = true;
                        ordered.push(card);
                    }
                });
                metrics.matched = ordered.length;
            }
            function finish() {
                if (completed) return;
                completed = true;
                clearTimeout(deadline);
                freeze();
                var more = states.some(function(st) {
                    return !st.exhausted && st.next <= st.total;
                });
                var pages = Math.max(1, Math.ceil(ordered.length / 20));
                if (more && ordered.length >= page * 20)
                    pages = Math.max(pages, page + 1);
                var results = ordered.slice((page - 1) * 20, page * 20);
                if (results.length) {
                    job.ok({ title: 'Новинки США · КП 7,5+',
                        url: KP8_CATALOG_URL, source: 'SURS',
                        page: page, total_pages: pages, results: results });
                } else if (job.fail) job.fail();
                working = false;
                run();
            }
            function next() {
                if (completed) return;
                // Count accepted cards, not raw API pages.
                if (cards.length >= page * 20) return finish();
                var todo = states.filter(function(st) {
                    return !st.exhausted && st.next <= st.total;
                });
                if (!todo.length) return finish();
                var pending = todo.length;
                todo.forEach(function(st) {
                    var n = st.next++;
                    metrics.pages_requested++;
                    kp8Fetch(st.type, n, ready, function(json) {
                        if (completed) return;
                        metrics.pages_completed++;
                        if (!json || !Array.isArray(json.results)) {
                            st.exhausted = true;
                            metrics.failed++;
                        } else {
                            var pages = Number(json.total_pages);
                            if (isFinite(pages) && pages > 0)
                                st.total = Math.min(st.total, Math.floor(pages));
                            if (n >= st.total || !json.results.length)
                                st.exhausted = true;
                            kp8Collect(json, st.type, seen, cards);
                        }
                        if (--pending === 0) next();
                    }, { from: st.year, to: st.year });
                });
            }
            if (ready) next();
            else kp8FindSource(function(found) {
                if (completed) return;
                ready = found;
                if (!ready) return finish();
                next();
            });
        }
        return {
            load: function(page, ok, fail) {
                queued.push({ page: Math.max(1, Number(page) || 1),
                    ok: ok, fail: fail });
                run();
            }
        };
    }

    // ContentRows.call может вернуть функцию, объект или массив функций/объектов.
    function resolveExternalPayload(value, callback) {
        var delivered = false;
        function finish(data) {
            if (delivered) return;
            delivered = true;
            callback(data);
        }
        if (value && typeof value.then === 'function') {
            try {
                value.then(function(data) { resolveExternalPayload(data, finish); }, function() {
                    finish({ results: [] });
                });
            } catch(e) {
                console.warn('[SURS] Ошибка Promise сторонней полки:', e);
                finish({ results: [] });
            }
            return;
        }
        if (typeof value === 'function') {
            try { value(function(data) { resolveExternalPayload(data, finish); }); }
            catch(e) { console.warn('[SURS] Ошибка функции ContentRow:', e); finish({ results: [] }); }
            return;
        }
        if (Array.isArray(value)) {
            var asRows = value.some(function(entry) {
                return typeof entry === 'function' || (entry && Array.isArray(entry.results));
            });
            if (!asRows) {
                finish({ results: value.filter(function(card) { return card && typeof card === 'object'; }) });
                return;
            }
            if (!value.length) { finish({ results: [] }); return; }
            var rows = new Array(value.length);
            var pending = value.length;
            value.forEach(function(entry, index) {
                var completed = false;
                resolveExternalPayload(entry, function(row) {
                    if (completed) return;
                    completed = true;
                    rows[index] = row && Array.isArray(row.results) ? row : { results: [] };
                    if (--pending === 0) {
                        var merged = { results: [] };
                        rows.forEach(function(part) {
                            if (!merged.title && part.title) merged.title = part.title;
                            merged.results = merged.results.concat(part.results);
                        });
                        finish(merged);
                    }
                });
            });
            return;
        }
        if (value && Array.isArray(value.results)) { finish(value); return; }
        finish({ results: [] });
    }

                owner.get('tv/top_rated', params, function(json) {
                    callback(prepareResults(json, 'Сериалы с высоким рейтингом'));
                }, function() { callback({ results: [] }); });
            };
        }

        if (shelfId === 'core_cartoons') {
            return function(callback) {
                var url = 'discover/movie?with_genres=16,10751&sort_by=popularity.desc&vote_count.gte=30';
                owner.get(url, params, function(json) {
                    callback(prepareResults(json, 'Семейные мультфильмы'));
                }, function() { callback({ results: [] }); });
            };
        }

        // ==========================================
        // ПОЛКИ TRAKT.TV (СМОТРЕТЬ ДАЛЬШЕ, WATCHLIST, РЕКОМЕНДАЦИИ, ТРЕНДЫ)
        // ==========================================
        if (shelfId.indexOf('trakt_') === 0 || shelfId === 'ext_TraktUpNextRow' || shelfId === 'ext_TraktWatchlistRow' || shelfId === 'ext_TraktRecommendationsRow' || shelfId === 'ext_TraktTrendingRow') {
            return function(callback) {
                var called = false;
                var timer = setTimeout(function() {
                    if (!called) {
                        called = true;
                        recordShelfStatus(shelfId, 'timeout');
                        console.warn('[SURS Trakt] Таймаут полки Trakt (3.5с):', shelfId);
                        callback({ results: [] });
                    }
                }, 3500);

                function normalizeTraktItem(raw) {
                    if (!raw) return null;
                    var media = raw.movie || raw.show || raw;
                    var ep = raw.episode || null;
                    var isTv = Boolean(raw.show || (raw.type === 'tv') || (media && media.first_aired));

                    var title = media.title || media.name || (ep ? ep.title : '') || 'Без названия';
                    var episodeNote = '';
                    if (ep && (ep.season != null || ep.number != null)) {
                        var sNum = (ep.season != null) ? (ep.season < 10 ? '0' + ep.season : '' + ep.season) : '01';
                        var eNum = (ep.number != null) ? (ep.number < 10 ? '0' + ep.number : '' + ep.number) : '01';
                        episodeNote = ' (s' + sNum + 'e' + eNum + (ep.title ? ' · ' + ep.title : '') + ')';
                    }

                    var year = media.year || (media.released ? ('' + media.released).slice(0, 4) : (media.first_aired ? ('' + media.first_aired).slice(0, 4) : ''));
                    var releaseDate = media.released || media.first_aired || (year ? year + '-01-01' : '');
                    var tmdbId = (media.ids && media.ids.tmdb) || raw.tmdb_id || media.tmdb_id || (media.source === 'tmdb' ? media.id : null);
                    if (!tmdbId) return null; // Без TMDB ID карточку нельзя корректно открыть.
                    var traktId = (media.ids && media.ids.trakt) || raw.trakt_id || media.trakt_id;
                    var imdbId = (media.ids && media.ids.imdb) || raw.imdb_id;

                    var posterPath = (media.images && media.images.poster && (media.images.poster.medium || media.images.poster.full)) || media.poster_path || raw.poster_path || '';
                    var backdropPath = (media.images && media.images.fanart && (media.images.fanart.medium || media.images.fanart.full)) || media.backdrop_path || raw.backdrop_path || '';

                    var card = {
                        source: 'tmdb',
                        method: isTv ? 'tv' : 'movie',
                        type: isTv ? 'tv' : 'movie',
                        id: tmdbId || traktId || raw.id,
                        tmdb_id: tmdbId,
                        trakt_id: traktId,
                        imdb_id: imdbId,
                        title: title + episodeNote,
                        name: title + episodeNote,
                        original_title: media.original_title || media.title || '',
                        original_name: media.original_name || media.name || '',
                        release_date: releaseDate,
                        first_air_date: releaseDate,
                        release_year: year ? ('' + year) : (releaseDate ? releaseDate.slice(0, 4) : ''),
                        vote_average: media.rating != null ? Math.round(media.rating * 10) / 10 : (raw.vote_average != null ? raw.vote_average : undefined),
                        overview: (ep && ep.overview) || media.overview || '',
                        poster_path: posterPath,
                        backdrop_path: backdropPath,
                        img: posterPath ? (posterPath.indexOf('http') === 0 ? posterPath : (Lampa.TMDB ? Lampa.TMDB.image('t/p/w300' + posterPath) : posterPath)) : '',
                        background_image: backdropPath ? (backdropPath.indexOf('http') === 0 ? backdropPath : (Lampa.TMDB ? Lampa.TMDB.image('t/p/w1280' + backdropPath) : backdropPath)) : ''
                    };

                    return card;
                }

                function enrichCardsWithTMDB(cards, onComplete) {
                    var needEnrich = cards.filter(function(c) {
                        return c && c.tmdb_id && !c.poster_path && !c.img;
                    });

                    if (!needEnrich.length || !Lampa.TMDB || !Lampa.TMDB.key || !Lampa.Network) {
                        return onComplete();
                    }

                    var count = Math.min(needEnrich.length, 6);
                    var done = 0;
                    var finished = false;
                    var tmdbTimer = setTimeout(function() {
                        if (!finished) {
                            finished = true;
                            onComplete();
                        }
                    }, 1200);

                    needEnrich.slice(0, 6).forEach(function(cardItem) {
                        var endpoint = (cardItem.type === 'tv' ? 'tv/' : 'movie/') + cardItem.tmdb_id + '?api_key=' + Lampa.TMDB.key() + '&language=ru';
                        var apiUrl = Lampa.TMDB.api(endpoint);
                        Lampa.Network.silent(apiUrl, function(data) {
                            if (data) {
                                if (data.poster_path) {
                                    cardItem.poster_path = data.poster_path;
                                    cardItem.img = Lampa.TMDB.image('t/p/w300' + data.poster_path);
                                }
                                if (data.backdrop_path) {
                                    cardItem.backdrop_path = data.backdrop_path;
                                    cardItem.background_image = Lampa.TMDB.image('t/p/w1280' + data.backdrop_path);
                                }
                                if (data.title || data.name) {
                                    var origTitle = cardItem.title;
                                    var baseTitle = data.title || data.name;
                                    if (origTitle.indexOf(' (s') > 0) {
                                        var epPart = origTitle.slice(origTitle.indexOf(' (s'));
                                        cardItem.title = baseTitle + epPart;
                                        cardItem.name = cardItem.title;
                                    } else {
                                        cardItem.title = baseTitle;
                                        cardItem.name = baseTitle;
                                    }
                                }
                                if (data.vote_average) cardItem.vote_average = Math.round(data.vote_average * 10) / 10;
                                if (data.release_date || data.first_air_date) {
                                    var rel = data.release_date || data.first_air_date;
                                    cardItem.release_date = rel;
                                    cardItem.first_air_date = rel;
                                    cardItem.release_year = ('' + rel).slice(0, 4);
                                }
                            }
                            done++;
                            if (done >= count && !finished) {
                                finished = true;
                                clearTimeout(tmdbTimer);
                                onComplete();
                            }
                        }, function() {
                            done++;
                            if (done >= count && !finished) {
                                finished = true;
                                clearTimeout(tmdbTimer);
                                onComplete();
                            }
                        }, false, { cache: { life: 60 * 24 * 3 } });
                    });
                }

                function finishTrakt(line) {
                    if (!called) {
                        called = true;
                        clearTimeout(timer);
                        if (!line) return callback({ results: [] });

                        var rawCards = Array.isArray(line) ? line : (line.results || line.items || []);
                        if (!Array.isArray(rawCards) || !rawCards.length) return callback({ results: [] });

                        var cards = rawCards.map(normalizeTraktItem).filter(Boolean);
                        if (!cards.length) return callback({ results: [] });

                        enrichCardsWithTMDB(cards, function() {
                            cards.forEach(function(item) {
                                if (!item.params) {
                                    item.params = {
                                        createInstance: function() { return Lampa.Maker.make('Card', item); },
                                        emit: {
                                            onlyEnter: function() { Lampa.Router.call('full', item); },
                                            onlyFocus: function() {
                                                if (Lampa.Background && Lampa.Utils && Lampa.Utils.cardImgBackgroundBlur) {
                                                    Lampa.Background.change(Lampa.Utils.cardImgBackgroundBlur(item));
                                                }
                                            }
                                        }
                                    };
                                }
                            });

                            var shelfTitle = line.title;
                            if (!shelfTitle) {
                                if (shelfId === 'trakt_continue' || shelfId === 'ext_TraktUpNextRow') shelfTitle = 'Trakt: Смотреть дальше';
                                else if (shelfId === 'trakt_watchlist' || shelfId === 'ext_TraktWatchlistRow') shelfTitle = 'Trakt: Хочу посмотреть';
                                else if (shelfId === 'trakt_recommendations' || shelfId === 'ext_TraktRecommendationsRow') shelfTitle = 'Trakt: Рекомендации';
                                else if (shelfId === 'trakt_trending' || shelfId === 'ext_TraktTrendingRow') shelfTitle = 'Trakt: В тренде';
                                else shelfTitle = 'Trakt.TV';
                            }

                            callback({
                                results: cards,
                                title: shelfTitle,
                                params: { items: { view: 20, mapping: 'line' } }
                            });
                        });
                    }
                }

                try {
                    var targetRowName = 'TraktUpNextRow';
                    var apiMethod = 'upnext';
                    if (shelfId === 'trakt_watchlist' || shelfId === 'ext_TraktWatchlistRow') {
                        targetRowName = 'TraktWatchlistRow';
                        apiMethod = 'watchlist';
                    } else if (shelfId === 'trakt_recommendations' || shelfId === 'ext_TraktRecommendationsRow') {
                        targetRowName = 'TraktRecommendationsRow';
                        apiMethod = 'recommendations';
                    } else if (shelfId === 'trakt_trending' || shelfId === 'ext_TraktTrendingRow') {
                        targetRowName = 'TraktTrendingRow';
                        apiMethod = 'trending';
                    }

                    // 1. Проверяем перехваченную ContentRows строку из плагина Trakt
                    var crShelf = window.SURS_EXTERNAL_SHELVES ? (window.SURS_EXTERNAL_SHELVES['ext_' + targetRowName] || window.SURS_EXTERNAL_SHELVES[shelfId]) : null;
                    if (crShelf && typeof crShelf.call === 'function') {
                        var nativeRow = crShelf.raw && typeof crShelf.raw.call === 'function' ? crShelf.raw : crShelf;
                        var resFunc = crShelf.call.call(nativeRow, params || {}, 'main');
                        if (resFunc != null) {
                            resolveExternalPayload(resFunc, finishTrakt);
                            return;
                        }
                    }

                    // 2. Проверяем глобальный API плагина Trakt (window.TraktTV.api)
                    if (typeof window !== 'undefined' && window.TraktTV && window.TraktTV.api) {
                        var api = window.TraktTV.api;
                        if (typeof api[apiMethod] === 'function') {
                            var response = api[apiMethod]({ limit: 20, page: 1 });
                            if (response && typeof response.then === 'function') {
                                response.then(finishTrakt, function() { finishTrakt(null); });
                            } else {
                                finishTrakt(response);
                            }
                            return;
                        }
                    }

                    // Native ContentRows has no public enumeration API.
                    // Do not run other plugin callbacks or inspect toString().
                    finishTrakt(null);
                } catch(e) {
                    console.error('[SURS Trakt] Ошибка загрузчика Trakt:', e);
                    finishTrakt(null);
                }
            };
        }

        // ==========================================
        // ВНЕШНИЕ СТОРОННИЕ ПОЛКИ И ПЛАГИНЫ (ПЕСОЧНИЦА С ТАЙМАУТОМ)
        // ==========================================
        var extShelf = window.SURS_EXTERNAL_SHELVES ? window.SURS_EXTERNAL_SHELVES[shelfId] : null;
        if (extShelf) {
            return function(callback) {
                var complete = false;
                var timer = setTimeout(function() {
                    if (complete) return;
                    complete = true;
                    recordShelfStatus(shelfId, 'timeout');
                    console.warn('[SURS] Внешняя полка превысила таймаут:', shelfId);
                    callback({ results: [] });
                }, shelfTimeout(shelfId) - 500);

                function finish(payload) {
                    if (complete) return;
                    complete = true;
                    clearTimeout(timer);
                    // A plugin may return null, a malformed value, or a frozen row.
                    if (!payload || !Array.isArray(payload.results)) {
                        payload = { results: [], title: extShelf.title || shelfId };
                    } else if (!payload.title) {
                        payload = Object.assign({}, payload, { title: extShelf.title || shelfId });
                    }
                    callback(payload);
                }

                try {
                    if (typeof extShelf.loader === 'function') {
                        extShelf.loader(function(data) {
                            resolveExternalPayload(data, finish);
                        }, params, owner);
                    } else if (typeof extShelf.call === 'function') {
                        // Match the native ContentRows receiver (row.call(params, screen)).
                        var nativeRow = extShelf.raw && typeof extShelf.raw.call === 'function' ? extShelf.raw : extShelf;
                        resolveExternalPayload(extShelf.call.call(nativeRow, params || {}, 'main'), finish);
                    } else {
                        // Полка сохранена, но исходный плагин пока не зарегистрирован.
                        finish({ results: [], title: extShelf.title || shelfId });
                    }
                } catch(err) {
                    console.warn('[SURS] Ошибка стороннего загрузчика:', shelfId, err);
                    finish({ results: [] });
                }
            };
        }

        return null;
    }

    // 6. ИСТОЧНИКИ ДАННЫХ TMDB (БЕЗ РАНДОМА)
    // ==========================================
    function createSourceTMDB(parent, mode) {
        return function() {
            this.network = new Lampa.Reguest();
            this.discovery = false;

            // Use the standard Lampa category_full screen and its native pagination.
            // Other categories keep the original TMDB implementation unchanged.
            this.list = function(listParams, onComplete, onError) {
                if (listParams && listParams.url === HOT_US_CATALOG_URL) {
                    hotCatalog(listParams.page || 1, this, listParams, onComplete, onError);
                    return;
                }
                if (listParams && listParams.url === KP8_CATALOG_URL) {
                if (!listParams._sursKP8Session) listParams._sursKP8Session = kp8CatalogSession();
                listParams._sursKP8Session.load(listParams.page || 1, onComplete, onError);
                return;
            }
            if (listParams && listParams.url === US8_CATALOG_URL) {
                    if (!listParams._sursUS8Session)
                        listParams._sursUS8Session = us8CatalogSession(this, listParams);
                    listParams._sursUS8Session.load(listParams.page || 1, onComplete, onError);
                    return;
                }
                if (typeof parent.list === 'function')
                    return parent.list(listParams, onComplete, onError);
                if (typeof onError === 'function') onError();
            };

            this.main = function(params, onComplete, onError) {
                params = params || {};
                var owner = this;

                // Never run ContentRows.call merely to discover rows.
                // It executes external handlers and cannot assign stable names.
                patchContentRows();

                var savedShelves = getSavedShelves(mode);
                var combinedData = [];
                if (!savedShelves.some(function(sh) { return sh.enabled; })) {
                    savedShelves = [getDefaultSectionShelves(mode)[0]];
                }

                // Only the active screen's independently saved profile determines
                // visibility and order; disabling a home shelf has no side effects.
                savedShelves.forEach(function(shelf) {
                    if (!shelf.enabled) return;
                    var fn = getShelfLoader(shelf.id, owner, params);
                    if (fn) {
                        fn.__sursId = shelf.id;
                        fn.__sursTimeout = shelfTimeout(shelf.id);
                        combinedData.push(fn);
                    }
                });

                if (!combinedData.length) {
                    var nav = getShelfLoader('nav_buttons', owner, params);
                    if (nav) { nav.__sursId = 'nav_buttons'; nav.__sursTimeout = shelfTimeout('nav_buttons'); combinedData.push(nav); }
                }

                // Защита от зависаний ТВ (Watchdog для каждой полки):
                // Если внешний плагин или TMDB задерживается - полка мирно отдает пустой массив,
                // не блокируя лоадер и не подвешивая интерфейс телевизора!
                var protectedData = combinedData.map(function(fn, i) {
                    return function(callback) {
                        var finished = false;
                        var shelfId = fn.__sursId;
                        recordShelfStatus(shelfId, 'loading');
                        var timer = setTimeout(function() {
                            if (!finished) {
                                finished = true;
                                recordShelfStatus(shelfId, 'timeout');
                                callback({ results: [] });
                            }
                        }, fn.__sursTimeout || 5000);

                        try {
                            fn(function(res) {
                                if (!finished) {
                                    finished = true;
                                    clearTimeout(timer);
                                    var old = window.SURS_SHELF_STATUS[shelfId];
                                    if (!old || old.state !== 'timeout') recordShelfStatus(shelfId, res && res.results && res.results.length ? 'ready' : 'empty');
                                    callback(res || { results: [] });
                                }
                            });
                        } catch(err) {
                            if (!finished) {
                                finished = true;
                                clearTimeout(timer);
                                recordShelfStatus(shelfId, 'error');
                                callback({ results: [] });
                            }
                        }
                    };
                });

                // Lampa 3: four concurrent loads, strict saved order and incremental
                // Main.onBuild additions. Never call Activity.toggle() per new row:
                // doing so restarts the TV controller and can reset remote focus.
                if (Lampa.Manifest && Number(Lampa.Manifest.app_digital) >= 300) {
                    window.SURS_LOAD_GENERATION = window.SURS_LOAD_GENERATION || {};
                    var generation = (window.SURS_LOAD_GENERATION[mode] || 0) + 1;
                    window.SURS_LOAD_GENERATION[mode] = generation;
                    var targetSource = mode === 'NEW' ? 'SURS NEW' :
                                       mode === 'RUS' ? 'SURS RUS' : 'SURS';
                    var metrics = { started_at: Date.now(), first_row_ms: null,
                                    completed_ms: null, total: protectedData.length, completed: 0,
                                    render_batches: 0, rendered_rows: 0 };
                    window.SURS_PERFORMANCE.screens[mode] = metrics;
                    var rows = protectedData.map(function() { return { done: false, data: null }; });
                    var queued = 0, emitted = 0, running = 0;
                    var pumping = false, flushPending = false, initialBuilt = false;
                    var deferred = [];

                    function currentScreen() {
                        if (window.SURS_LOAD_GENERATION[mode] !== generation) return false;
                        var active = Lampa.Activity && Lampa.Activity.active && Lampa.Activity.active();
                        return !active || typeof active !== 'object' || !active.component ||
                               (active.component === 'main' && active.source === targetSource);
                    }
                    function publish(batch) {
                        if (!batch.length || !currentScreen()) return;
                        metrics.render_batches++;
                        metrics.rendered_rows += batch.length;
                        if (metrics.first_row_ms === null)
                            metrics.first_row_ms = Date.now() - metrics.started_at;
                        if (!initialBuilt) {
                            initialBuilt = true;
                            onComplete(batch);
                            return;
                        }
                        var active = Lampa.Activity && Lampa.Activity.active && Lampa.Activity.active();
                        var component = active && active.activity && active.activity.component;
                        if (component && typeof component.emit === 'function') {
                            component.emit('build', batch);
                        } else {
                            deferred = deferred.concat(batch);
                        }
                    }
                    function flushReady() {
                        flushPending = false;
                        if (!currentScreen()) return;
                        var batch = [];
                        while (emitted < rows.length && rows[emitted].done) {
                            var data = rows[emitted].data;
                            if (data && Array.isArray(data.results) && data.results.length) batch.push(data);
                            emitted++;
                        }
                        if (batch.length) publish(batch);
                        if (emitted === rows.length) {
                            metrics.completed_ms = Date.now() - metrics.started_at;
                            if (!initialBuilt) onError();
                        }
                    }
                    function scheduleFlush() {
                        if (flushPending) return;
                        flushPending = true;
                        setTimeout(flushReady, 0);
                    }
                    function pump() {
                        if (pumping || !currentScreen()) return;
                        pumping = true;
                        while (running < 4 && queued < protectedData.length) {
                            (function(index) {
                                queued++;
                                running++;
                                var done = false;
                                function finish(data) {
                                    if (done) return;
                                    done = true;
                                    running--;
                                    rows[index].done = true;
                                    rows[index].data = data;
                                    metrics.completed++;
                                    scheduleFlush();
                                    if (!pumping) pump();
                                }
                                try { protectedData[index](finish); }
                                catch(e) { finish({ results: [] }); }
                            })(queued);
                        }
                        pumping = false;
                    }
                    // During onCreate, Activity.push has not pushed the new activity yet.
                    setTimeout(pump, 0);
                    return function(resolve, reject) {
                        if (deferred.length) resolve(deferred.splice(0, deferred.length));
                        else if (typeof reject === 'function') reject();
                    };
                }

                // Lampa 2: preserve the existing partNext loading behavior.
                var isFirstBatch = true;
                function loadPart(partLoaded, partEmpty) {
                    var chunkLimit = isFirstBatch ? 6 : 4;
                    isFirstBatch = false;
                    Lampa.Api.partNext(protectedData, chunkLimit, partLoaded, partEmpty);
                }

                loadPart(onComplete, onError);
                return loadPart;
            };
        };
    }

    // ==========================================
    // 7. РЕГИСТРАЦИЯ И ЗАПУСК ПЛАГИНА
    // ==========================================
    // Lampa's Activity.init schedules its initial page after 500ms. A network
    // extension can register after that page, so rescue ONLY a mismatched
    // default TMDB main, never a user-selected view or a deep link.
    var sursBootRecoveryDone = false, sursBootRecoveryBound = false;
    var sursBootRecoveryStarted = 0;
    function ensureSursActiveOnBoot() {
        if (sursBootRecoveryDone || !window.plugin_surs_fixed_ready ||
            Date.now() - sursBootRecoveryStarted > 8000) return;
        try {
            var active = Lampa.Activity && Lampa.Activity.active && Lampa.Activity.active();
            if (!active || typeof active !== 'object' || !active.component) return;
            if (active.component !== 'main') {
                sursBootRecoveryDone = true;
                return;
            }
            if (active.source === 'SURS' || active.source === 'SURS NEW' || active.source === 'SURS RUS') {
                sursBootRecoveryDone = true;
                return;
            }
            if (active.source !== 'tmdb' && active.source !== 'surs_mod') {
                sursBootRecoveryDone = true;
                return;
            }
            if (window.start_deep_link ||
                (Lampa.Storage && Lampa.Storage.field &&
                 Lampa.Storage.field('start_page') === 'last')) {
                sursBootRecoveryDone = true;
                return;
            }
            if (!Lampa.Storage || Lampa.Storage.get('source') !== 'SURS') return;
            sursBootRecoveryDone = true;
            window.SURS_PERFORMANCE.boot_rescue = {
                from: active.source, to: 'SURS', at: Date.now()
            };
            refreshSursHome('MAIN');
        } catch(err) { console.warn('[SURS] Cold boot recovery failed:', err); }
    }
    function scheduleSursBootRecovery() {
        sursBootRecoveryStarted = Date.now();
        if (!sursBootRecoveryBound) {
            sursBootRecoveryBound = true;
            if (Lampa.Listener && Lampa.Listener.follow) {
                Lampa.Listener.follow('activity', function(e) {
                    if (e && e.type === 'start' && !sursBootRecoveryDone)
                        setTimeout(ensureSursActiveOnBoot, 0);
                });
            }
        }
        setTimeout(ensureSursActiveOnBoot, 0);
        setTimeout(ensureSursActiveOnBoot, 650);
    }

    function startPlugin() {
        if (window.plugin_surs_fixed_ready || window.plugin_surs_fixed_starting) return;
        var initStartedAt = Date.now();
        if (typeof Lampa === 'undefined' || !Lampa.Api || !Lampa.Api.sources || !Lampa.Api.sources.tmdb) {
            if (window.surs_retry_count++ < 66) {
                setTimeout(startPlugin, 150);
            } else {
                console.error('SURS: не удалось дождаться источника TMDB');
            }
            return;
        }
        window.plugin_surs_fixed_starting = true;
        try {
            addPluginStyles();

        var TMDBMain = createSourceTMDB(Lampa.Api.sources.tmdb, 'MAIN');
        var TMDBNew = createSourceTMDB(Lampa.Api.sources.tmdb, 'NEW');
        var TMDBRus = createSourceTMDB(Lampa.Api.sources.tmdb, 'RUS');

        var surs_mod = Object.assign({}, Lampa.Api.sources.tmdb, new TMDBMain());
        var surs_mod_new = Object.assign({}, Lampa.Api.sources.tmdb, new TMDBNew());
        var surs_mod_rus = Object.assign({}, Lampa.Api.sources.tmdb, new TMDBRus());

        Lampa.Api.sources.SURS = surs_mod;
        Lampa.Api.sources['SURS NEW'] = surs_mod_new;
        Lampa.Api.sources['SURS RUS'] = surs_mod_rus;
        Lampa.Api.sources.surs_mod = surs_mod;

        // Если текущий источник не настроен или стандартный, переключаем на SURS
        var curSource = Lampa.Storage.get('source');
        if (!curSource || curSource === 'tmdb' || curSource === 'surs_mod') {
            Lampa.Storage.set('source', 'SURS');
        }

        // Интеграция в настройки Lampa
        if (Lampa.SettingsApi) {
            Lampa.SettingsApi.addComponent({
                component: 'surs',
                name: 'SURS: Главный экран',
                icon: '<svg height="24" width="24" viewBox="0 0 24 24" fill="#ffffff"><path d="M3 6h18v2H3V6m0 5h18v2H3v-2m0 5h18v2H3v-2Z"/></svg>'
            });

            Lampa.SettingsApi.addParam({
                component: 'surs',
                param: { name: 'surs_open_editor', type: 'button' },
                field: {
                    name: 'Настройка полок Главного экрана',
                    description: 'Изменить порядок полок (▲/▼), скрыть лишние или добавить новые'
                },
                onChange: function() {
                    openShelfManagerModal('MAIN');
                }
            });

            Lampa.SettingsApi.addParam({
                component: 'surs',
                param: {name:'surs_hot_diagnostics',type:'button'},
                field: {name:'Диагностика «Горячие новинки США»',
                    description:'Публикация GitHub Actions, источники, кандидаты, актуальность'},
                onChange:openHotDiagnostics
            });

            Lampa.SettingsApi.addParam({
                component: 'surs',
                param: { name: 'surs_open_world_editor', type: 'button' },
                field: {
                    name: 'Настроить «Новинки Мир»',
                    description: 'Отдельные полки мировых новинок, независимо от главной'
                },
                onChange: function() { openShelfManagerModal('NEW'); }
            });
            Lampa.SettingsApi.addParam({
                component: 'surs',
                param: { name: 'surs_open_russia_editor', type: 'button' },
                field: {
                    name: 'Настроить «Новинки Россия»',
                    description: 'Отдельные российские фильмы и сериалы, не меняя главную'
                },
                onChange: function() { openShelfManagerModal('RUS'); }
            });

            Lampa.SettingsApi.addParam({
                component: 'surs',
                param: { type: 'title' },
                field: {
                    name: 'Кнопки быстрого перехода'
                }
            });

            SURS_NAV_SETTINGS.forEach(function(option) {
                Lampa.SettingsApi.addParam({
                    component: 'surs',
                    param: {
                        name: navButtonSettingKey(option.id),
                        type: 'trigger',
                        default: true
                    },
                    field: {
                        name: 'Показывать: ' + option.title,
                        description: 'Отображение кнопки на главном экране SURS'
                    },
                    onChange: function() {
                        // Lampa сохраняет trigger. Не закрываем настройки при
                        // каждом нажатии, чтобы можно было изменить сразу несколько.
                        if (Lampa.Noty && typeof Lampa.Noty.show === 'function') {
                            Lampa.Noty.show('Кнопки сохранены. Откройте главную SURS для обновления.');
                        }
                    }
                });
            });

            Lampa.SettingsApi.addParam({
                component: 'surs',
                param: { name: 'surs_nav_refresh', type: 'button' },
                field: {
                    name: 'Применить расположение кнопок',
                    description: 'Обновить главный экран после изменения видимости'
                },
                onChange: function() {
                    refreshSursHome();
                }
            });

            Lampa.SettingsApi.addParam({
                component: 'surs',
                param: { name: 'surs_reset_order', type: 'button' },
                field: {
                    name: 'Сбросить порядок по умолчанию',
                    description: 'Вернуть исходную структуру полок'
                },
                onChange: function() {
                    Lampa.Select.show({
                        title: 'Сбросить полки SURS по умолчанию?',
                        items: [
                            { title: 'Да, сбросить и отключить сторонние полки', action: 'confirm' },
                            { title: 'Отмена', action: 'cancel' }
                        ],
                        onSelect: function(choice) {
                            if (choice.action === 'confirm') {
                                if (executeBulletproofReset()) {
                                    Lampa.Noty.show('Восстановлены 9 активных полок из 23');
                                    refreshSursHome();
                                }
                            }
                        },
                        onBack: function() { Lampa.Controller.toggle('settings'); }
                    });
                }
            });
        }

        // Запуск перехвата и загрузка ранее сохраненных исследованных полок (если они были сохранены до сброса)
        patchContentRows();
        loadDiscoveredFromStorage();

        // Адаптация бокового Главного меню Lampa (ТВ, телефон, планшет, браузер)
        function setupMainMenuIntegration() {
            try {
                var menuList = $('.menu .menu__list').eq(0);
                if (!menuList.length) return;

                var mainButton = $('.menu__item[data-action="main"]');
                // Сохраняем штатный обработчик главного меню Lampa.
                if (mainButton.length && !mainButton.hasClass('surs-customized')) {
                    mainButton.addClass('surs-customized');
                }

                // Добавление быстрого пункта "Порядок полок" в меню
                if (!$('.menu__item[data-action="surs-manager"]').length) {
                    var managerBtn = $('<li class="menu__item selector" data-action="surs-manager">' +
                        '<div class="menu__ico">' + buttonIcons.surs_shelf_manager + '</div>' +
                        '<div class="menu__text">Порядок полок</div>' +
                        '</li>');
                    managerBtn.on('hover:enter click', function(e) {
                        if (e && e.stopPropagation) e.stopPropagation();
                        openShelfManagerModal();
                    });
                    var settingsBtn = $('.menu__item[data-action="settings"]');
                    if (settingsBtn.length) {
                        settingsBtn.before(managerBtn);
                    } else {
                        menuList.append(managerBtn);
                    }
                }
            } catch (e) {
                console.error('SURS: Ошибка в setupMainMenuIntegration', e);
            }
        }

        setTimeout(setupMainMenuIntegration, 400);
        Lampa.Listener.follow('activity', function(e) {
            if (e.type === 'start') setTimeout(setupMainMenuIntegration, 300);
        });
        Lampa.Listener.follow('profile', function(e) {
            if (e.type === 'changed') setTimeout(setupMainMenuIntegration, 300);
        });

            // Announce readiness only after all registrations and listeners succeed.
            window.plugin_surs_fixed_ready = true;
            window.surs_retry_count = 0;
            window.SURS_PERFORMANCE.ready_at = Date.now();
            window.SURS_PERFORMANCE.init_ms = Date.now() - initStartedAt;
            scheduleSursBootRecovery();
            kp8PreloadEnabledSource();
            console.log('SURS Pro v4.0.8 (Hot Discovery 4.5 interest) успешно инициализирован.');
        } catch(err) {
            console.error('[SURS] Ошибка запуска, повторная попытка:', err);
            if (window.surs_retry_count++ < 10) {
                setTimeout(startPlugin, 1000);
            } else {
                console.error('[SURS] Не удалось завершить инициализацию');
            }
        } finally {
            window.plugin_surs_fixed_starting = false;
        }
    }

    // Экспорт для быстрого вызова из консоли или других плагинов
    window.openSursShelfManager = openShelfManagerModal;
    window.LampaShelfManager = {
        open: openShelfManagerModal,
        register: registerDiscoveredShelf,
        rescan: function(notify) { return runAutoDiscovery(notify !== false); },
        getAllShelves: getAllAvailableShelves,
        getSavedShelves: getSavedShelves,
        saveShelves: saveShelves,
        resetDefaults: function(scope) {
            var name = normalizeShelfScope(scope);
            var defaults = executeBulletproofReset(name);
            if (defaults) refreshSursHome(name);
            return defaults;
        },
        getDefaultSectionShelves: getDefaultSectionShelves,
        hotDiagnostics: function(){return window.SURS_PERFORMANCE.hot_us||{};},
        openHotDiagnostics: openHotDiagnostics,
        version: '4.0.8'
    };

    // Запуск плагина с поддержкой как ранней, так и поздней загрузки
    if (window.appready || (typeof Lampa !== 'undefined' && Lampa.Activity && Lampa.Activity.active && Lampa.Activity.active())) {
        startPlugin();
    } else if (typeof Lampa !== 'undefined' && Lampa.Listener && typeof Lampa.Listener.follow === 'function') {
        Lampa.Listener.follow('app', function (e) {
            if (e.type === 'ready') startPlugin();
        });
    } else {
        // The Lampa global may not exist yet when an extension script is evaluated.
        startPlugin();
    }
})();
